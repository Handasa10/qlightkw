// ===== Long hall with a gypsum cove: one LED strip along both sides, showing voltage drop (12V) vs even light (24V) ===== // x: -2..2, floor y=0, raised ceiling 3.2, perimeter drop 2.85; the hall runs from z=2 (camera end) to z=-14 const HW = 2.0, Z0 = 1.6, Z1 = -13.6, CEIL_H = 3.2, DROP_H = 2.85; marbleTex.repeat.set(4, 16); const floor = plane(4, 16, M.floor, [0, 0, -6], [-Math.PI / 2, 0, 0]); const wallM = new THREE.MeshStandardMaterial({ map: plasterTex, color: '#e8dfd2', roughness: .9 }); box(.1, CEIL_H, 16.4, wallM, -HW - .1, 0, -14.2); box(.1, CEIL_H, 16.4, wallM, HW, 0, -14.2); box(4.2, CEIL_H, .1, wallM, -2.1, 0, -14.2); box(4.2, CEIL_H, .1, wallM, -2.1, 0, 2.2); const gyp = new THREE.MeshStandardMaterial({ color: '#f4f0ea', roughness: .95 }); box(4, .06, 16.4, gyp, -2, CEIL_H, -14.2, { cast: false }); // raised ceiling for (const sx of [-1, 1]) { // perimeter drop with a cove lip box(.55, .1, 16.4, gyp, sx < 0 ? -HW : HW - .55, DROP_H - .1, -14.2); box(.04, .12, 16.4, gyp, sx < 0 ? -HW + .55 : HW - .59, DROP_H - .02, -14.2); // lip hiding the strip } box(4, .1, .5, gyp, -2, DROP_H - .1, -14.2); box(4, .1, .4, gyp, -2, DROP_H - .1, 1.8); // far wall feature: fluted wood panel + niche box(2.2, 2.4, .06, M.fluted, -1.1, .2, -14.1); // majlis seating along both walls const seat = new THREE.MeshStandardMaterial({ map: fabricTex, color: '#d9c7a6', roughness: .95 }); const back = new THREE.MeshStandardMaterial({ map: fabricTex, color: '#8a6a4a', roughness: .95 }); for (const sx of [-1, 1]) { box(.7, .38, 11, seat, sx < 0 ? -HW + .02 : HW - .72, 0, -12.5, { r: .05 }); box(.18, .5, 11, back, sx < 0 ? -HW + .02 : HW - .2, .38, -12.5, { r: .05 }); for (let i = 0; i < 9; i++) box(.12, .32, .4, M.goldCush, sx < 0 ? -HW + .2 : HW - .32, .42, -12 + i * 1.2, { r: .05 }); } plane(2.2, 10, M.rug, [0, .006, -7], [-Math.PI / 2, 0, 0]); for (const z of [-4, -8.5]) { const t = new THREE.Mesh(new THREE.CylinderGeometry(.4, .4, .35, 48), M.darkWood); t.position.set(0, .175, z); t.castShadow = true; scene.add(t); } // the strip: segments along each cove. The wash on the raised ceiling is an additive gradient band per segment (cheap and // continuous); a few long rect lights add the bounce into the room. Each segment's brightness follows the voltage profile. const SEG = 28, SL = (Z0 - Z1) / SEG; const washTex = canvasTex(64, 256, (g, w, h) => { const gr = g.createLinearGradient(0, 0, 0, h); gr.addColorStop(0, 'rgba(255,255,255,1)'); gr.addColorStop(.25, 'rgba(255,255,255,.55)'); gr.addColorStop(1, 'rgba(255,255,255,0)'); g.fillStyle = gr; g.fillRect(0, 0, w, h); }, [1, 1], false); const strip = []; // { m, w, t, sx } for (const sx of [-1, 1]) for (let i = 0; i < SEG; i++) { const z = Z0 - (i + .5) * SL, x = sx * (HW - .62); const m = new THREE.Mesh(new THREE.BoxGeometry(.03, .012, SL * .995), emis('#000', 0)); m.position.set(x, DROP_H + .02, z); scene.add(m); const wm = new THREE.MeshBasicMaterial({ map: washTex, color: '#000', transparent: true, blending: THREE.AdditiveBlending, depthWrite: false, toneMapped: false }); const w = new THREE.Mesh(new THREE.PlaneGeometry(SL, 1.25), wm); w.rotation.x = Math.PI / 2; w.rotation.z = Math.PI / 2 * (sx < 0 ? 1 : -1); w.position.set(x - sx * .55, CEIL_H - .006, z); scene.add(w); // the band also climbs the cove's inner face a little const v = new THREE.Mesh(new THREE.PlaneGeometry(SL, .35), wm); v.position.set(sx * (HW - .55 + .001) - sx * .0, DROP_H + .17, z); v.rotation.y = sx < 0 ? Math.PI / 2 : -Math.PI / 2; scene.add(v); strip.push({ m, wm, t: i / (SEG - 1), sx }); } const bounce = []; for (const sx of [-1, 1]) for (let i = 0; i < 6; i++) { const len = (Z0 - Z1) / 6, z = Z0 - (i + .5) * len, x = sx * (HW - .9); const l = new THREE.RectAreaLight('#ffd6a0', 0, .5, len); l.position.set(x, CEIL_H - .05, z); l.lookAt(x, 0, z); scene.add(l); bounce.push({ l, t: i / 5 }); } const hemi = new THREE.HemisphereLight('#ffe9d2', '#3a2c20', .05); scene.add(hemi); const fillL = new THREE.PointLight('#ffcf9a', 0, 10, 2); fillL.position.set(0, 1.4, -13); scene.add(fillL); const VIEWS = { h: { pos: [0, 1.45, 1.9], tgt: [0, 2.15, -12], fov: 72 }, f: { pos: [0, 1.6, -5.2], tgt: [0, 2.55, -13.4], fov: 62 }, u: { pos: [.5, 1.55, -1.2], tgt: [-1.3, 2.9, -4.2], fov: 62 }, }; const C12 = new THREE.Color('#ffd6a0'), CLOW = new THREE.Color('#ff9a4a'); // state: off | 12v | 24v window.setState = (st) => { const prof = (t) => st === '12v' ? 1 - .93 * Math.pow(t, .95) : st === '24v' ? 1 - .05 * t : 0; for (const s of strip) { const k = prof(s.t), col = C12.clone(); if (st === '12v') col.lerp(CLOW, Math.pow(s.t, 1.1)); s.wm.color.copy(col).multiplyScalar(.85 * k); s.m.material = k > 0 ? emis('#' + col.getHexString(), 3 * k + .3) : emis('#000', 0); } for (const b of bounce) { b.l.intensity = 2.2 * prof(b.t); b.l.color.copy(st === '12v' ? C12.clone().lerp(CLOW, b.t * .7) : C12); } fillL.intensity = st === 'off' ? .8 : 1.2; hemi.intensity = st === 'off' ? .15 : .12; scene.environmentIntensity = .04; renderer.toneMappingExposure = 1.05; };
